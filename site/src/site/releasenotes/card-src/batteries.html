@HttpBuilderClient('https://api.example.com')
interface UserApi {
    @Get('/users/{id}')
    User getUser(String id)        // path param
    @Get('/users')
    List<User> search(String name)  // ?name=...
    @Post('/users')
    User create(@Body Map user)    // JSON body
}

def user = UserApi.create().getUser('42')
